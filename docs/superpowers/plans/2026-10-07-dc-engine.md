## Terminology

Agreed with Terence 2026-10-08, used in code, comments, and conversation.

| Instant | Name |
|---|---|
| 0 | proposal time |
| Δ | vote time |
| 2Δ | confirmation cutoff, a vote must be resolved by now to support a confirmation |
| 3Δ | support cutoff, a vote must be received by now to count for the next slot's vote |
| 6Δ | evaluation time, the next slot's confirmation cutoff, when the slot-s confirmation is evaluated |

Terms: vote, blockVote (votes carried in a block body), participants and supported (the two halves of a view), received (vote), processed (block), target / justified / finalized pairs, viable (FG height filter), eligible (relative-majority gate), equivocation, live confirmed, latest confirmed, AC, FG. Lean names stay only inside `// Spec:` comments. Stabilization is a TODO, not designed.

## Review focus

Inputs the spec implies but no trace in the handoff table pins down. Each line names the owning task and the test added there.

1. A vote whose head is a later-slot block than the vote's slot stays raw and never supports (Task 4, `TestResolveRejectsLaterSlotHead`).
2. A proposal whose support list contains a vote not in its raw list, or a carried vote of the wrong slot, is rejected before admission (Task 5, `TestOnBlockRejectsMalformedCarriedVotes`).
3. An equivocating proposer contributes two disjoint carried sets, so the merged raw view can hold more than two votes per validator and manufacture an equivocator. The engine must not cap the merged view (Task 6, `TestVoterViewUnionsAllSlotProposals`).
4. `h_max - 1` truncates at zero. With `h_max <= 1` every block is viable and the height clause of eligibility is false for every block (Task 7, `TestViableTruncatesAtZero`).
5. A clock jump across several slots must evaluate every skipped slot's confirmation before the pool prune removes its votes, and the pool for slots `s-1` and `s` survives the prune (Task 3, `TestTickKeepsPreviousSlotPool`, and Task 9, `TestTickDispatchesConfirmationOnce`).

```go
package decoupled

import (
	"math"

	"github.com/OffchainLabs/prysm/v7/consensus-types/primitives"
)

// Spec: Time, Stamp. Units are the caller's, Δ scales the schedule.
type Time int64

const genesisStamp Time = math.MinInt64

type Config struct {
	Delta Time
}

// Spec: GoldfishVote (val_index, slot, head).
type Vote struct {
	Validator primitives.ValidatorIndex
	Slot      primitives.Slot
	Head      [32]byte
}

type Pair struct {
	Height primitives.Height
	Root   [32]byte
}

// Spec: the fields of σ[B] the fork choice reads. Entry is zero while the deferred root fill is pending.
type FGSummary struct {
	Height         primitives.Height
	Entry          [32]byte
	Nonjustifiable bool
	Justified      Pair
	Finalized      Pair
}

type Node struct {
	Root      [32]byte
	Slot      primitives.Slot
	Parent    *Node
	Children  []*Node
	Votes     []Vote
	Support   []Vote
	FG        FGSummary
	depth     int
	processed Time
}

type poolEntry struct {
	vote Vote
	at   Time
}

// Spec: Store Σ, with the SG fields removed and stabilization stubbed.
type Store struct {
	cfg              Config
	t                Time
	s                primitives.Slot
	root             *Node
	nodes            map[[32]byte]*Node
	pool             map[primitives.Slot][]poolEntry
	finalized        *Node
	justified        *Node
	hJ               primitives.Height
	hMax             primitives.Height
	liveConfirmed    *Node
	latestConfirmed  *Node
	confirmedThrough primitives.Slot
	hasConfirmed     bool
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

// Spec: Store.init. Genesis carries ChainState.initial, so h_max starts at 1.
func New(cfg Config, genesisRoot [32]byte) *Store {
	g := &Node{
		Root:      genesisRoot,
		processed: genesisStamp,
		FG:        FGSummary{Height: 1, Entry: genesisRoot, Justified: Pair{Root: genesisRoot}, Finalized: Pair{Root: genesisRoot}},
	}
	return &Store{
		cfg:             cfg,
		root:            g,
		nodes:           map[[32]byte]*Node{genesisRoot: g},
		pool:            make(map[primitives.Slot][]poolEntry),
		finalized:       g,
		justified:       g,
		hMax:            1,
		liveConfirmed:   g,
		latestConfirmed: g,
	}
}

func (st *Store) Node(root [32]byte) *Node { return st.nodes[root] }

func (st *Store) HasNode(root [32]byte) bool { return st.nodes[root] != nil }

func (st *Store) link(root [32]byte, parent *Node, slot primitives.Slot) *Node {
	n := &Node{Root: root, Slot: slot, Parent: parent, depth: parent.depth + 1}
	parent.Children = append(parent.Children, n)
	st.nodes[root] = n
	return n
}

// Spec: Block.preceq, a ⪯ b.
func isAncestor(a, b *Node) bool {
	for b != nil && b.depth > a.depth {
		b = b.Parent
	}
	return a == b
}

func (st *Store) IsAncestor(ancestor, descendant [32]byte) bool {
	a, b := st.nodes[ancestor], st.nodes[descendant]
	return a != nil && b != nil && isAncestor(a, b)
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

// Spec: slotStart, t_s = 4Δs.
func (st *Store) slotStart(s primitives.Slot) Time {
	return 4 * st.cfg.Delta * Time(s)
}

// Spec: slotOfTime, floor division clamped at slot 0.
func (st *Store) slotOf(t Time) primitives.Slot {
	if t < 0 {
		return 0
	}
	return primitives.Slot(t / (4 * st.cfg.Delta))
}

func (st *Store) voteTime(s primitives.Slot) Time         { return st.slotStart(s) + st.cfg.Delta }
func (st *Store) supportCutoff(s primitives.Slot) Time    { return st.slotStart(s) + 2*st.cfg.Delta }
func (st *Store) viewFreeze(s primitives.Slot) Time       { return st.slotStart(s) + 3*st.cfg.Delta }
func (st *Store) confirmationTime(s primitives.Slot) Time { return st.slotStart(s) + 6*st.cfg.Delta }

func (st *Store) Slot() primitives.Slot { return st.s }

// Slot s-1 is still read by the slot-s voter view and the slot-(s-1) confirmation, older slots are not.
func (st *Store) OnTick(now Time) {
	st.t = now
	st.s = st.slotOf(now)
	for k := range st.pool {
		if k+1 < st.s {
			delete(st.pool, k)
		}
	}
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

func countVotes(votes []Vote, val primitives.ValidatorIndex) int {
	n := 0
	for _, v := range votes {
		if v.Validator == val {
			n++
		}
	}
	return n
}

// Spec: equivocates, two distinct votes by one validator in the set.
func equivocates(votes []Vote, val primitives.ValidatorIndex) bool {
	return countVotes(votes, val) >= 2
}

func participates(votes []Vote, val primitives.ValidatorIndex) bool {
	return countVotes(votes, val) >= 1
}

func dedupe(votes []Vote) []Vote {
	seen := make(map[Vote]struct{}, len(votes))
	out := votes[:0:0]
	for _, v := range votes {
		if _, ok := seen[v]; ok {
			continue
		}
		seen[v] = struct{}{}
		out = append(out, v)
	}
	return out
}

func (st *Store) poolVotes(k primitives.Slot) []Vote {
	entries := st.pool[k]
	out := make([]Vote, len(entries))
	for i, e := range entries {
		out[i] = e.vote
	}
	return out
}

func (st *Store) stampOf(v Vote) (Time, bool) {
	for _, e := range st.pool[v.Slot] {
		if e.vote == v {
			return e.at, true
		}
	}
	return 0, false
}

// Spec: on_goldfish_vote. The slot window is [s-1, s], duplicates and a third distinct vote are dropped unstamped.
func (st *Store) OnVote(v Vote, now Time) bool {
	if v.Slot+1 < st.s || st.s < v.Slot {
		return false
	}
	if _, dup := st.stampOf(v); dup {
		return false
	}
	if equivocates(st.poolVotes(v.Slot), v.Validator) {
		return false
	}
	st.pool[v.Slot] = append(st.pool[v.Slot], poolEntry{vote: v, at: now})
	return true
}

// Spec: resolved. A head from a later slot than the vote never resolves.
func (st *Store) resolve(v Vote) *Node {
	h := st.nodes[v.Head]
	if h == nil || h.Slot > v.Slot {
		return nil
	}
	return h
}

// Spec: resolution_time τ = max(stamp(vote), stamp(head)), unresolved otherwise.
func (st *Store) tau(v Vote) (Time, bool) {
	at, ok := st.stampOf(v)
	if !ok {
		return 0, false
	}
	h := st.resolve(v)
	if h == nil {
		return 0, false
	}
	return max(at, h.processed), true
}
```

```go
package decoupled

import (
	"bytes"
	"errors"

	"github.com/OffchainLabs/prysm/v7/consensus-types/primitives"
)

var (
	ErrFutureBlock            = errors.New("block slot is ahead of the store slot")
	ErrUnknownParent          = errors.New("block parent is not in the store")
	ErrNotFinalizedDescendant = errors.New("block does not descend from the finalized block")
	ErrParentSlot             = errors.New("block slot is not after its parent slot")
	ErrCarriedVotes           = errors.New("carried votes are malformed")
)

type Block struct {
	Root    [32]byte
	Parent  [32]byte
	Slot    primitives.Slot
	Votes   []Vote
	Support []Vote
	FG      FGSummary
}

// Spec: carried_votes_well_formed and carried_support_well_formed.
func carriedVotesWellFormed(b Block) bool {
	carried := make(map[Vote]struct{}, len(b.Votes))
	for _, v := range b.Votes {
		if v.Slot+1 != b.Slot {
			return false
		}
		carried[v] = struct{}{}
	}
	for _, v := range b.Support {
		if _, ok := carried[v]; !ok {
			return false
		}
	}
	return true
}

// Spec: on_block. The proposer check is Prysm's block validation, not the store's.
func (st *Store) OnBlock(b Block, now Time) error {
	if st.s < b.Slot {
		return ErrFutureBlock
	}
	if st.nodes[b.Root] != nil {
		return nil
	}
	parent := st.nodes[b.Parent]
	if parent == nil {
		return ErrUnknownParent
	}
	if !isAncestor(st.finalized, parent) {
		return ErrNotFinalizedDescendant
	}
	if parent.Slot >= b.Slot {
		return ErrParentSlot
	}
	if !carriedVotesWellFormed(b) {
		return ErrCarriedVotes
	}
	n := st.link(b.Root, parent, b.Slot)
	n.processed = now
	n.Votes = b.Votes
	n.Support = b.Support
	n.FG = b.FG
	for _, v := range b.Votes {
		st.OnVote(v, now)
	}
	st.updateFinality(n)
	return nil
}

// Spec: HeightId order, (h_j, J.root) lexicographic with the greatest root winning.
func pairLess(aH primitives.Height, aR [32]byte, bH primitives.Height, bR [32]byte) bool {
	if aH != bH {
		return aH < bH
	}
	return bytes.Compare(aR[:], bR[:]) < 0
}

// Spec: update_finality. Three statements in order, each reading the store the previous one wrote.
func (st *Store) updateFinality(n *Node) {
	st.hMax = max(st.hMax, n.FG.Height)
	j := st.nodes[n.FG.Justified.Root]
	if j != nil && isAncestor(st.finalized, j) && pairLess(st.hJ, st.justified.Root, n.FG.Justified.Height, j.Root) {
		st.justified, st.hJ = j, n.FG.Justified.Height
	}
	f := st.nodes[n.FG.Finalized.Root]
	if f != nil && f != st.finalized && isAncestor(st.finalized, f) && isAncestor(f, st.justified) && st.viableInFullTree(f) {
		st.finalized = f
		st.prune()
	}
}

// Spec: viable with the witness ranging over the whole processed tree. Replaced by Task 7's viable.
func (st *Store) viableInFullTree(b *Node) bool {
	stack := []*Node{b}
	for len(stack) > 0 {
		n := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if n.FG.Height+1 >= st.hMax {
			return true
		}
		stack = append(stack, n.Children...)
	}
	return false
}

func (st *Store) Finalized() *Node                 { return st.finalized }
func (st *Store) Justified() *Node                 { return st.justified }
func (st *Store) JustifiedHeight() primitives.Height { return st.hJ }
func (st *Store) MaxHeight() primitives.Height       { return st.hMax }
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

type view struct {
	raw     []Vote
	support []Vote
}

func (st *Store) slotBlocks(s primitives.Slot) []*Node {
	var out []*Node
	for _, n := range st.nodes {
		if n.Slot == s {
			out = append(out, n)
		}
	}
	return out
}

// Spec: voter_view and voter_support_view for the slot-s vote. Carried support is the proposer's fixed view, only head availability is rechecked.
func (st *Store) voterView(s primitives.Slot) view {
	if s == 0 {
		return view{}
	}
	prev := s - 1
	freeze := st.viewFreeze(prev)
	var v view
	for _, e := range st.pool[prev] {
		if e.at < freeze {
			v.raw = append(v.raw, e.vote)
		}
		if tau, ok := st.tau(e.vote); ok && tau < freeze {
			v.support = append(v.support, e.vote)
		}
	}
	for _, b := range st.slotBlocks(s) {
		for _, u := range b.Votes {
			if u.Slot == prev {
				v.raw = append(v.raw, u)
			}
		}
		for _, u := range b.Support {
			if u.Slot == prev && st.resolve(u) != nil {
				v.support = append(v.support, u)
			}
		}
	}
	v.raw = dedupe(v.raw)
	v.support = dedupe(v.support)
	return v
}

// Spec: voter_processed_block_tree. Blocks stamped before the previous freeze plus every ancestor of a slot-s proposal.
func (st *Store) voterBlocks(s primitives.Slot) map[*Node]bool {
	in := make(map[*Node]bool, len(st.nodes))
	if s == 0 {
		return in
	}
	freeze := st.viewFreeze(s - 1)
	for _, n := range st.nodes {
		if n.processed < freeze {
			in[n] = true
		}
	}
	for _, p := range st.slotBlocks(s) {
		for n := p; n != nil && !in[n]; n = n.Parent {
			in[n] = true
		}
	}
	return in
}

// Spec: proposer_view and proposer_support_view, the whole previous-slot bucket with no freeze.
func (st *Store) proposerView(s primitives.Slot) view {
	if s == 0 {
		return view{}
	}
	var v view
	for _, e := range st.pool[s-1] {
		v.raw = append(v.raw, e.vote)
		if st.resolve(e.vote) != nil {
			v.support = append(v.support, e.vote)
		}
	}
	return v
}
```

```go
package decoupled

import (
	"bytes"

	"github.com/OffchainLabs/prysm/v7/consensus-types/primitives"
)

type scores struct {
	equivocators int
	participants int
	subtree      map[*Node]int
}

// Spec: goldfish_score and voters_count. Support is attributed by walking from the resolved head up to the root once per clean supporter.
func (st *Store) score(v view, s primitives.Slot) *scores {
	sc := &scores{subtree: make(map[*Node]int)}
	counts := make(map[primitives.ValidatorIndex]int)
	for _, u := range v.raw {
		counts[u.Validator]++
	}
	sc.participants = len(counts)
	for _, c := range counts {
		if c >= 2 {
			sc.equivocators++
		}
	}
	for _, u := range v.support {
		if u.Slot != s || counts[u.Validator] >= 2 {
			continue
		}
		for n := st.resolve(u); n != nil; n = n.Parent {
			sc.subtree[n]++
		}
	}
	return sc
}

func (sc *scores) of(b *Node) int { return sc.equivocators + sc.subtree[b] }

// Spec: the relative-majority gate, voters_count < 2 * goldfish_score.
func (sc *scores) majority(b *Node) bool { return sc.participants < 2*sc.of(b) }

// Spec: outranks, higher score then greatest root.
func outranks(score func(*Node) int, b, c *Node) bool {
	sb, sc := score(b), score(c)
	return sb > sc || (sb == sc && bytes.Compare(b.Root[:], c.Root[:]) > 0)
}

// Spec: ghost. Membership in the tree is tested on the child only, so the anchor may sit outside it.
func ghost(anchor *Node, in func(*Node) bool, score func(*Node) int, eligible func(*Node) bool) *Node {
	h := anchor
	for {
		var best *Node
		for _, c := range h.Children {
			if !in(c) || !eligible(c) {
				continue
			}
			if best == nil || outranks(score, c, best) {
				best = c
			}
		}
		if best == nil {
			return h
		}
		h = best
	}
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

func all(*Node) bool { return true }

// Spec: viable. The allowed views are ancestor closed because a block is processed only after its parent, so subtrees under an excluded node hold no witness.
func (st *Store) viable(b *Node, in func(*Node) bool) bool {
	bound := st.hMax
	if bound > 0 {
		bound--
	}
	stack := []*Node{b}
	for len(stack) > 0 {
		n := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if !in(n) {
			continue
		}
		if n.FG.Height >= bound {
			return true
		}
		stack = append(stack, n.Children...)
	}
	return false
}

// Spec: get_fg_root. J when h_max = h_j + 1, otherwise F.
func (st *Store) fgRoot() *Node {
	if st.hMax == st.hJ+1 {
		return st.justified
	}
	return st.finalized
}

// Spec: get_filtered_block_tree_from. Viability and membership are both recomputed inside the supplied view.
func (st *Store) filteredTree(blocks func(*Node) bool) func(*Node) bool {
	root := st.fgRoot()
	memo := make(map[*Node]bool)
	return func(b *Node) bool {
		if v, ok := memo[b]; ok {
			return v
		}
		v := blocks(b) && isAncestor(st.finalized, b) && isAncestor(root, b) && st.viable(b, blocks)
		memo[b] = v
		return v
	}
}

// Spec: goldfish_eligible. The majority gate applies only inside the band {h_max-1, h_max}.
func (st *Store) eligible(sc *scores, cur primitives.Slot) func(*Node) bool {
	bound := st.hMax
	if bound > 0 {
		bound--
	}
	return func(b *Node) bool {
		parentHeight := b.FG.Height
		if b.Parent != nil {
			parentHeight = b.Parent.FG.Height
		}
		return parentHeight < bound || sc.majority(b) || b.Slot == cur
	}
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

type Proposal struct {
	Head    *Node
	Votes   []Vote
	Support []Vote
}

func (st *Store) headIn(blocks func(*Node) bool, v view, k primitives.Slot) *Node {
	sc := st.score(v, k)
	return ghost(st.anchor(), st.filteredTree(blocks), sc.of, st.eligible(sc, st.s))
}

// Spec: with_proposal_input. The proposer view is the whole previous-slot bucket, the walk is over the full tree.
func (st *Store) ProposalInputs(s primitives.Slot) Proposal {
	v := st.proposerView(s)
	return Proposal{Head: st.headIn(all, v, s-1), Votes: v.Raw(), Support: v.Support()}
}

// Spec: goldfish_vote. Only the vote duty restricts the walk to the voter's frozen processed-block view.
func (st *Store) VoteHead(s primitives.Slot) *Node {
	blocks := st.voterBlocks(s)
	return st.headIn(func(n *Node) bool { return blocks[n] }, st.voterView(s), s-1)
}

// Before the vote instant nobody has voted in this slot, so the previous slot's full bucket is the freshest evidence. After it, this is the head read of get_fg_vote.
func (st *Store) Head() *Node {
	if st.t < st.voteTime(st.s) {
		return st.headIn(all, st.proposerView(st.s), st.s-1)
	}
	raw := st.poolVotes(st.s)
	v := view{raw: raw}
	for _, u := range raw {
		if st.resolve(u) != nil {
			v.support = append(v.support, u)
		}
	}
	return st.headIn(all, v, st.s)
}

func (st *Store) LiveConfirmed() *Node { return st.liveConfirmed }
```

```go
// Spec: floor_on_stable.
func floorOnStable(stable, candidate, old *Node) *Node {
	if isAncestor(stable, candidate) {
		return candidate
	}
	if isAncestor(stable, old) {
		return old
	}

// Spec: update_confirmation_with. Early support uses τ before 2Δ, the denominator and the second-vote test use receipts before 6Δ.
func (st *Store) UpdateConfirmation(s primitives.Slot) {
	early := st.supportCutoff(s)
	late := st.confirmationTime(s)
	var lateVotes, support []Vote
	for _, e := range st.pool[s] {
		if e.at < late {
			lateVotes = append(lateVotes, e.vote)
		}
	}
	for _, e := range st.pool[s] {
		if tau, ok := st.tau(e.vote); ok && tau < early && countVotes(lateVotes, e.vote.Validator) < 2 {
			support = append(support, e.vote)
		}
	}
	// Spec passes support_votes as both score arguments and voters_count over late_votes.
	sc := st.score(view{raw: support, support: support}, s)
	seen := make(map[primitives.ValidatorIndex]struct{}, len(lateVotes))
	for _, u := range lateVotes {
		seen[u.Validator] = struct{}{}
	}
	sc.participants = len(seen)
	h := ghost(st.anchor(), st.filteredTree(all), sc.of, sc.majority)
	if sc.majority(h) {
		st.liveConfirmed = h
		st.latestConfirmed = floorOnStable(st.stable(), advanceConfirmed(st.latestConfirmed, h), st.latestConfirmed)
		return
	}
	st.liveConfirmed = st.fgRoot()
	st.latestConfirmed = floorOnStable(st.stable(), st.latestConfirmed, st.latestConfirmed)
}

func (st *Store) LatestConfirmed() *Node { return st.latestConfirmed }

// Spec: get_confirmed over get_stable, with the stable record stubbed to F.
func (st *Store) Confirmed() *Node {
	if isAncestor(st.stable(), st.latestConfirmed) {
		return st.latestConfirmed
	}
	return st.stable()
}
```

```go
package decoupled

import "github.com/OffchainLabs/prysm/v7/consensus-types/primitives"

type HeightPairKind uint8

const (
	EmptyPair HeightPairKind = iota
	TimeoutPair
	TargetPair
)

// Spec: NamedHeightPair. Target holds the entry for a timeout too, as the wire format does.
type HeightPair struct {
	Kind   HeightPairKind
	Height primitives.Height
	Target [32]byte
}

type FinalityPair struct {
	Set    bool
	Height primitives.Height
	Target [32]byte
}

// Spec: the (h_c, T_c, ν) fields read from the source block's state. The Bool is the nj bit.
type Source struct {
	Set            bool
	Height         primitives.Height
	Entry          [32]byte
	Nonjustifiable bool
}

// Spec: Record Λ. Lives with the validator, must survive reorgs and restarts.
type Record struct {
	targets  map[primitives.Height][32]byte
	timeouts map[primitives.Height]bool
	locks    map[primitives.Height][32]byte
}

func NewRecord() *Record {
	return &Record{
		targets:  make(map[primitives.Height][32]byte),
		timeouts: make(map[primitives.Height]bool),
		locks:    make(map[primitives.Height][32]byte),
	}
}

// Spec: finality_pair.
func (r *Record) finalityPair(hJ primitives.Height, j [32]byte, hF primitives.Height) FinalityPair {
	if hF >= hJ {
		return FinalityPair{}
	}
	if t, ok := r.targets[hJ]; ok && t != j {
		return FinalityPair{}
	}
	if r.timeouts[hJ] {
		return FinalityPair{}
	}
	if l, ok := r.locks[hJ]; ok && l != j {
		return FinalityPair{}
	}
	return FinalityPair{Set: true, Height: hJ, Target: j}
}

// Spec: own_lock. This attestation's finality pair overrides the recorded lock at its height so one attestation never forms E1 evidence.
func (r *Record) ownLock(h primitives.Height, fp FinalityPair) ([32]byte, bool) {
	if fp.Set && fp.Height == h {
		return fp.Target, true
	}
	l, ok := r.locks[h]
	return l, ok
}

// Spec: height_pair, the seven rows in order.
func (r *Record) heightPair(src Source, fp FinalityPair) HeightPair {
	if !src.Set {
		return HeightPair{Kind: EmptyPair}
	}
	timeout := HeightPair{Kind: TimeoutPair, Height: src.Height, Target: src.Entry}
	target := HeightPair{Kind: TargetPair, Height: src.Height, Target: src.Entry}
	if r.timeouts[src.Height] {
		return timeout
	}
	if locked, ok := r.ownLock(src.Height, fp); ok {
		if locked == src.Entry {
			return target
		}
		return HeightPair{Kind: EmptyPair}
	}
	if recorded, ok := r.targets[src.Height]; ok {
		if recorded == src.Entry {
			return target
		}
		return timeout
	}
	if src.Nonjustifiable {
		return timeout
	}
	return target
}

// Spec: record_attestation, the only writer of Λ.
func (r *Record) record(hp HeightPair, fp FinalityPair) {
	if fp.Set {
		r.locks[fp.Height] = fp.Target
	}
	switch hp.Kind {
	case TimeoutPair:
		r.timeouts[hp.Height] = true
	case TargetPair:
		if _, ok := r.targets[hp.Height]; !ok {
			r.targets[hp.Height] = hp.Target
		}
	}
}

// Spec: create_attestation. Record, then release.
func (r *Record) CreateAttestation(src Source, hJ primitives.Height, j [32]byte, hF primitives.Height) (HeightPair, FinalityPair) {
	fp := r.finalityPair(hJ, j, hF)
	hp := r.heightPair(src, fp)
	r.record(hp, fp)
	return hp, fp
}
```
